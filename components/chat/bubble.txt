<template>
  <div :class="['flex flex-col', isUser ? 'items-end' : 'items-start']">
    <div class="flex items-start relative">
      <!-- Avatar -->
      <div
        v-if="!isUser"
        class="w-8 h-8 rounded-full overflow-hidden mr-3 flex-shrink-0"
      >
        <img :src="avatar" alt="Avatar" class="w-full h-full object-cover" />
      </div>

      <!-- Message Bubble -->
      <div
        :class="[
          'rounded-lg py-2 px-4 max-w-xs relative group',
          isUser ? 'bg-blue-500 text-white' : 'bg-gray-100 text-gray-700',
        ]"
      >
        <!-- Message Content -->
        <div v-if="!isUser && isThinking" class="thinking-dots">
          <span>.</span><span>.</span><span>.</span>
        </div>
        <p v-else-if="!isUser" class="text-[14px]">{{ displayedText }}</p>
        <p v-else class="text-[14px] text-white">{{ text }}</p>

        <!-- Action Buttons Container -->
        <div
          v-if="!isUser && !isThinking"
          :class="[
            'absolute flex items-center gap-2 transition-opacity opacity-0 group-hover:opacity-100',
            'bottom-[-36px] left-0 right-0 justify-start', // Adjusted bottom value
          ]"
        >
          <!-- All Buttons in One Row -->
          <div class="flex gap-2">
            <button class="chat-tool-btn" @click="$emit('toggleSpeak', text)">
              <div class="sign"><icon-voice /></div>
              <div class="text">{{ speakIsAudio ? "停止" : "朗读" }}</div>
            </button>
            <button
              class="chat-tool-btn"
              @click="$emit('refreshAnswer', question)"
            >
              <div class="sign"><icon-refresh /></div>
              <div class="text">重新生成</div>
            </button>
            <button class="chat-tool-btn" @click="handleCopy">
              <div class="sign"><icon-copy /></div>
              <div class="text">复制答案</div>
            </button>
            <button class="chat-tool-btn" @click="$emit('exportWord', text)">
              <div class="sign"><icon-share-internal /></div>
              <div class="text">导出Word</div>
            </button>
            <button class="chat-tool-btn" @click="$emit('exportPdf', text)">
              <div class="sign"><icon-download /></div>
              <div class="text">导出PDF</div>
            </button>
            <a-popconfirm
              content="此操作将永久删除当条消息,确定要删除吗?"
              @confirm="$emit('delete', messageId)"
              type="warning"
            >
              <button class="chat-tool-btn">
                <div class="sign"><icon-delete /></div>
                <div class="text">删除</div>
              </button>
            </a-popconfirm>
            <!-- Thumbs Up Button -->
            <button class="chat-tool-btn" @click="handleThumbsUp">
              <div class="sign">
                <svg
                  class="w-4 h-4"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <path
                    stroke-linecap="round"
                    stroke-linejoin="round"
                    stroke-width="2"
                    d="M14 10h4.764a2 2 0 011.789 2.894l-3.5 7A2 2 0 0115.263 21h-4.017c-.163 0-.326-.02-.485-.06L7 20m7-10V5a2 2 0 00-2-2h-.095c-.5 0-.905.405-.905.905 0 .714-.211 1.412-.608 2.006L7 11v9m7-10h-2M7 20H5a2 2 0 01-2-2v-6a2 2 0 012-2h2.5"
                  />
                </svg>
              </div>
              <div class="text">好评</div>
            </button>
            <!-- Thumbs Down Button -->
            <button class="chat-tool-btn" @click="handleThumbsDown">
              <div class="sign">
                <svg
                  class="w-4 h-4"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <g transform="rotate(180 12 12)">
                    <path
                      stroke-linecap="round"
                      stroke-linejoin="round"
                      stroke-width="2"
                      d="M14 10h4.764a2 2 0 011.789 2.894l-3.5 7A2 2 0 0115.263 21h-4.017c-.163 0-.326-.02-.485-.06L7 20m7-10V5a2 2 0 00-2-2h-.095c-.5 0-.905.405-.905.905 0 .714-.211 1.412-.608 2.006L7 11v9m7-10h-2M7 20H5a2 2 0 01-2-2v-6a2 2 0 012-2h2.5"
                    />
                  </g>
                </svg>
              </div>
              <div class="text">差评</div>
            </button>
          </div>
        </div>
      </div>
    </div>
  </div>
</template>

<script>
import { ref, watch, onMounted, onUnmounted } from "vue";
import {
  IconVoice,
  IconRefresh,
  IconCopy,
  IconShareInternal,
  IconDownload,
  IconDelete,
} from "@arco-design/web-vue/es/icon";
import { Popconfirm } from "@arco-design/web-vue";

export default {
  name: "MessageBubble",
  components: {
    IconVoice,
    IconRefresh,
    IconCopy,
    IconShareInternal,
    IconDownload,
    IconDelete,
    Popconfirm,
  },
  props: {
    text: { type: String, required: true },
    timestamp: { type: Date, required: true },
    isUser: { type: Boolean, required: true },
    avatar: { type: String, required: true },
    messageId: { type: [String, Number], default: null },
    isThinking: { type: Boolean, default: false },
    typingSpeed: { type: Number, default: 1 },
    initialDelay: { type: Number, default: 10 },
    question: { type: String, default: "" }, // Added for refreshAnswer
    speakIsAudio: { type: Boolean, default: false }, // Added for speak button state
  },
  setup(props, { emit }) {
    const displayedText = ref("");
    let intervalId = null;

    const applyEffects = () => {
      if (intervalId) {
        clearInterval(intervalId);
        intervalId = null;
      }
      if (props.isUser) {
        displayedText.value = props.text;
        return;
      }
      if (props.isThinking) {
        displayedText.value = "";
        return;
      }
      displayedText.value = "";
      setTimeout(() => {
        let index = 0;
        intervalId = setInterval(() => {
          if (index < props.text.length) {
            displayedText.value += props.text[index];
            index++;
          } else {
            clearInterval(intervalId);
            intervalId = null;
          }
        }, props.typingSpeed);
      }, props.initialDelay);
    };

    onMounted(applyEffects);
    watch(() => [props.text, props.isThinking], applyEffects);
    onUnmounted(() => {
      if (intervalId) clearInterval(intervalId);
    });

    const handleThumbsUp = () => {
      emit("feedback", { type: "positive", messageId: props.messageId });
    };

    const handleThumbsDown = () => {
      emit("feedback", { type: "negative", messageId: props.messageId });
    };

    const handleCopy = () => {
      navigator.clipboard.writeText(props.text).then(
        () => console.log("Copied to clipboard"),
        (err) => console.error("Failed to copy:", err)
      );
    };

    return {
      displayedText,
      handleThumbsUp,
      handleThumbsDown,
      handleCopy,
    };
  },
};
</script>
